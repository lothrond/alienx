# Gaming/performance tunables, applied to every host in inventory.
# Previously role defaults (roles/gaming_performance/defaults/main.yml) —
# moved here per request so they're visible/overridable in one place
# alongside any other host vars, rather than buried in role internals.
#
# Override per-target by adding a more specific file, e.g.
# group_vars/console.yml or host_vars/<hostname>.yml — those will take
# precedence over this file for hosts in that group.

# --- CPU ---
gaming_cpu_governor_performance: true      # cpupower governor -> performance
gaming_irqbalance_enabled: false           # ArchWiki: irqbalance can cause
                                            # stutter/bad framerates in games.
                                            # Left off for console; flip true
                                            # for base/desktop server-style use.

# --- GameMode ---
gaming_gamemode_enabled: true
gaming_gamemode_ioprio: 0                  # 0 = high priority (best-effort class)
gaming_gamemode_renice: 10                 # nice -10 while GameMode is active
gaming_gamemode_softrealtime: "auto"       # "auto" | "on" | "off"
gaming_gamemode_inhibit_screensaver: true

# --- NVIDIA (Maxwell / nvidia-tesla-470 branch) ---
gaming_nvidia_powermizer_max_perf: true    # force PowerMizer to max-perf level
gaming_nvidia_enable_msi: true             # MSI interrupts, usually a perf win;
                                            # flip false first if you hit the
                                            # stability issues Debian's wiki warns about
gaming_nvidia_xorg_tuning: true            # RenderAccel/TripleBuffer/MigrationHeuristic
gaming_nvidia_coolbits: 0                  # 0 disables overclocking bits;
                                            # set 28 (or 12) only if you intend
                                            # to overclock via nvidia-settings

# --- Kernel / sysctl (not yet wired into tasks/main.yml — see note below) ---
gaming_sysctl_vm_max_map_count: true       # some Proton/native titles need this raised
gaming_sysctl_vm_max_map_count_value: 2147483642
gaming_sysctl_vm_swappiness: true
gaming_sysctl_vm_swappiness_value: 10
gaming_sysctl_vm_dirty_tuning: false       # off by default; mainly a spinning-
                                            # disk / heavy-write-workload knob,
                                            # limited upside on an NVMe console box
gaming_disable_nmi_watchdog: false         # small overhead reduction; costs you
                                            # a hang-diagnosis tool, opt-in only

# --- Storage / I/O scheduler (not yet wired into tasks/main.yml) ---
gaming_io_scheduler_tuning: true
gaming_io_scheduler_nvme: "none"           # NVMe has its own queueing; let it work
gaming_io_scheduler_ssd: "mq-deadline"
gaming_io_scheduler_hdd: "bfq"             # BFQ helps responsiveness on spinning disks

# --- Vulkan / Mesa (not yet wired into tasks/main.yml) ---

# --- Overlay / diagnostics (not yet wired into tasks/main.yml) ---
gaming_mangohud_enabled: true
